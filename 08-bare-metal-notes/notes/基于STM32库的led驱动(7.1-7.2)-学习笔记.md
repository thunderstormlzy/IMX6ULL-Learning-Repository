# 基于STM32库的led驱动(7.1-7.2)

BSS清零是C语言运行时的"环境净化"操作，它保证所有全局变量和静态变量从0开始，而不是从随机垃圾值开始，确保程序逻辑的确定性。这是启动代码（_start）必须完成的核心任务之一。
start.s的内容
.global _start              /* 声明 _start 为全局符号，链接器入口点 */
/* ============================================================
 * 【新增部分】声明外部符号（链接脚本中定义的）
 * ============================================================ */
.global __bss_start         /* 声明 __bss_start 为全局符号，由链接脚本提供 */
                            /* 表示 BSS 段在内存中的起始地址 */
.global __bss_end           /* 声明 __bss_end 为全局符号，由链接脚本提供 */
                            /* 表示 BSS 段在内存中的结束地址 */
_start:
    /* 1. 将处理器调整为SVC模式 */
    mrs r0, cpsr            /* 读取当前程序状态寄存器到 r0 */
    bic r0, r0, #0x1f       /* 清除低 5 位（处理器模式位） */
    orr r0, r0, #0x13       /* 设置低 5 位为 0x13（SVC 管理员模式） */
    msr cpsr, r0            /* 写回 CPSR，切换到 SVC 模式 */
    /* ============================================================
     * 【新增部分】2. 清除BSS段
     * ============================================================ */
    ldr r0, =__bss_start    /* 将 BSS 段起始地址加载到 r0 */
                            /* __bss_start 由链接脚本（imx6u.lds）定义 */
    ldr r1, =__bss_end      /* 将 BSS 段结束地址加载到 r1 */
                            /* __bss_end 由链接脚本（imx6u.lds）定义 */
    mov r2, #0              /* 将 r2 设置为 0，用于清零操作 */
bss_loop:                   /* BSS 段清零循环入口 */
    cmp r0, r1              /* 比较当前地址（r0）是否到达结束地址（r1） */
    bhs bss_done            /* 如果 r0 >= r1，跳转到 bss_done（清除完成） */
                            /* bhs = Branch if Higher or Same（无符号数比较） */
    str r2, [r0], #4        /* 将 r2（0）存储到 r0 指向的地址 */
                            /* [r0], #4 表示：先存储，然后 r0 = r0 + 4（后索引） */
                            /* 相当于：*r0 = 0; r0 += 4; */
    b bss_loop              /* 无条件跳转回 bss_loop，继续循环 */
bss_done:                   /* BSS 段清零完成 */
    /* ============================================================
     * 【原有部分】设置栈指针并跳转到 main
     * ============================================================ */
    ldr sp, =0x80200000     /* 设置栈指针为 0x80200000（DDR 内存地址） */
    b main                  /* 跳转到 C 语言的 main 函数 */
相比于单纯的使用Makefile来编译和链接程序，加入了链接脚本来更加精准的控制每个不同类型段的内存地址
imx6u.lds
/* ============================================================
 * 链接脚本：自定义链接脚本
 * 目标平台：NXP i.MX6ULL / 正点原子 I.MX6U 开发板
 * 内存起始地址：0x87800000（DDR 内存，用于裸机程序）
 * ============================================================ */
SECTIONS {
    /* ============================================================
     * 设置当前地址计数器为 0x87800000
     * 即：后续所有段从 0x87800000 开始存放
     * 0x87800000 是正点原子 I.MX6U 开发板裸机程序的默认链接地址
     * ============================================================ */
    . = 0X87800000;
    /* ============================================================
     * 1. 代码段（.text）
     * ============================================================ */
    .text : {
        /* 首先放置 start.o 中的 .text 段 */
        /* start.o 是启动文件（start.s 编译得到），必须放在最前面 */
        /* 因为程序入口 _start 在 start.s 中，需要确保入口代码在起始地址 */
        start.o(.text)
        /* 然后放置所有其他目标文件（包括 main.o）中的 .text 段 */
        *(.text)
    }
    /* ============================================================
     * 2. 只读数据段（.rodata）
     * ALIGN(4) 表示 4 字节对齐
     * 存放：字符串常量、const 修饰的只读变量等
     * ============================================================ */
    .rodata ALIGN(4) : {
        *(.rodata*)          /* * 通配符，匹配所有 .rodata 开头的段 */
    }
    /* ============================================================
     * 3. 数据段（.data）
     * ALIGN(4) 表示 4 字节对齐
     * 存放：已初始化（非 0）的全局变量和静态变量
     * 例如：int a = 10; static int b = 20;
     * ============================================================ */
    .data ALIGN(4) : {
        *(.data*)            /* 匹配所有 .data 开头的段 */
    }
    /* ============================================================
     * 4. BSS 段起始地址标记
     * __bss_start 变量会被记录在符号表中，可在汇编代码中使用
     * start.s 中用 ldr r0, =__bss_start 获取此地址
     * ============================================================ */
    __bss_start = .;
    /* ============================================================
     * 5. BSS 段（.bss）
     * ALIGN(4) 表示 4 字节对齐
     * 存放：未初始化或初始化为 0 的全局变量和静态变量
     * 例如：int a; static int b; int c = 0;
     * 注意：BSS 段在程序中不占用文件空间，但运行时需要清零
     * ============================================================ */
    .bss ALIGN(4) : {
        *(.bss)              /* 匹配所有目标文件中的 .bss 段 */
        *(COMMON)            /* 匹配所有目标文件中的 COMMON 段（未初始化的全局变量） */
    }
    /* ============================================================
     * 6. BSS 段结束地址标记
     * __bss_end 变量记录 BSS 段结束地址
     * start.s 中用 ldr r1, =__bss_end 获取此地址
     * 然后从 __bss_start 到 __bss_end 循环清零
     * ============================================================ */
    __bss_end = .;
}
