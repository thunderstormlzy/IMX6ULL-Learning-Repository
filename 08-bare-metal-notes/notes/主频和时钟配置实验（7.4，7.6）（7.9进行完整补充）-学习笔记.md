# 主频和时钟配置实验（7.4，7.6）（7.9进行完整补充）

主要流程：先切到安全时钟源 -> 配 PLL/PFD -> 等 LOCK/STABLE ->配分频（看情况是否需要等待握手）->切换到所需时钟源- > 等握手 -> 切回原时钟。
若某个分频器在 CCM/CDHIPR 中有对应 busy 位，则修改后也要等待 busy 清零；本实验主要等待的是 选择器的握手。
重点：在选择器选择时，应该注意是否存在握手机制（即同时供给时钟切换模块（同步MUX）与MMDC（内存管理/控制单元）的频率，在时钟模块进行更改时，向MMDC发送请求信号，直到MMDC更改才会正式选择）

在main.c中的初始化优先级较高，防止时钟频率出现问题。
void clk_init(void)
{
    /*PLL1初始化AMR的主频初始*/
    if((((CCM->CCSR) >> 2)&0x1)==0)
    {
        CCM->CCSR &= ~(1<<8);
        CCM->CCSR |= (1<<2);
    }
    CCM_ANALOG->PLL_ARM =((66<<0)&(0X7F))|(1<<13);
     while((CCM_ANALOG->PLL_ARM & (1 << 31)) == 0); /*等待PLL1锁定，因为ARM需要使用*/
    CCM->CACRR = 1;
    CCM->CCSR &= ~(1<<2);
    /*PLL2初始化USB主频以及其PFD0-3*/
    unsigned int init=0;
    init = CCM_ANALOG->PFD_528;
    init&=~(0X3F3F3F3F);        /*清零*/
        init |= 32<<24;				/* PLL2_PFD3=528*18/32=297Mhz 	*/
	init |= 24<<16;				/* PLL2_PFD2=528*18/24=396Mhz(DDR使用的时钟，最大400Mhz) */
	init |= 16<<8;				/* PLL2_PFD1=528*18/16=594Mhz 	*/
	init |= 27<<0;				/* PLL2_PFD0=528*18/27=352Mhz  	*/
	CCM_ANALOG->PFD_528=init;
     while((CCM_ANALOG->PFD_528 & (1 << 22)) == 0); /*等待PLL2锁定，因为下面的AHB需要使用*/
    /*PLL3初始化主频及其分路*/
    init =0;
    init = CCM_ANALOG->PFD_480;
    init&=~(0X3F3F3F3F);        /*清零*/
	init |= 19<<24;				/* PLL3_PFD3=480*18/19=454.74Mhz 	*/
	init |= 17<<16;				/* PLL3_PFD2=480*18/17=508.24Mhz 	*/
	init |= 16<<8;				/* PLL3_PFD1=480*18/16=540Mhz		*/
	init |= 12<<0;				/* PLL3_PFD0=480*18/12=720Mhz	 	*/
	CCM_ANALOG->PFD_480=init;	/* 设置PLL3_PFD0~3 					*/	
    /*AHB初始化，来供给EPIT-I2C和ADC-WDOG*/
    CCM->CBCMR &= ~(3 << 18); 	/* 清除设置*/ 
    CCM->CBCMR |= (1 << 18);	/* pre_periph_clk=PLL2_PFD2=396MHz */
    CCM->CBCDR &= ~(1 << 25);	/* periph_clk=pre_periph_clk=396MHz */
    while(CCM->CDHIPR & (1 << 5));/* 等待握手完成 */
    /*AHB的位修改需要禁止AHB的输出，
    但不知道AHB的输出寄存器，
    因此使用AHB的默认3分频，
    AHB_ROOT_CLK等于396/3=132Mhz。*/
    /*配置CBCDR(IPG_PODF)控制的分频器分频为2， IPG_CLK_ROOT=132/2=66Mhz*/
    /*CSCMR1控制的选择器选择IPG_CLK_ROOT,分频器分频为1,PER_CLK_ROOT=66/1=66Mhz*/
    CCM->CBCDR &= ~(3 << 8); 	/* 清除设置*/
    CCM->CBCDR |= (1 << 8);		/* IPG_PODF=2 */
    CCM->CSCMR1 &= ~(1 << 6); 	/* 选择器选择IPG_CLK_ROOT*/
    CCM->CSCMR1 &= ~(63<< 0);	/* 分频器分频为1 */
}
